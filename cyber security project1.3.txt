<script>

function detectPhishing() {

    let email = document
        .getElementById("emailInput")
        .value
        .toLowerCase();

    let suspiciousKeywords = [

        "urgent",
        "verify",
        "password",
        "bank",
        "account",
        "login",
        "click",
        "free",
        "winner",
        "prize",
        "claim",
        "immediately",
        "suspended",
        "blocked",
        "confirm",
        "update your account",
        "limited time",
        "lottery"

    ];

    let suspiciousFound = [];

    let score = 0;


    // Check suspicious keywords

    suspiciousKeywords.forEach(function(keyword) {

        if (email.includes(keyword)) {

            suspiciousFound.push(keyword);
            score++;

        }

    });


    // Check URLs

    let urlPattern =
        /(https?:\/\/[^\s]+|www\.[^\s]+)/g;

    let urls = email.match(urlPattern);

    if (urls) {

        score += urls.length;

    }


    // Get result elements

    let result =
        document.getElementById("result");

    let resultTitle =
        document.getElementById("resultTitle");

    let resultText =
        document.getElementById("resultText");

    let featureList =
        document.getElementById("featureList");


    result.style.display = "block";

    featureList.innerHTML = "";


    // Display suspicious keywords

    if (suspiciousFound.length > 0) {

        let item =
            document.createElement("li");

        item.innerHTML =
            "<strong>Suspicious Keywords:</strong> "
            + suspiciousFound.join(", ");

        featureList.appendChild(item);

    }

    else {

        let item =
            document.createElement("li");

        item.innerHTML =
            "No suspicious keywords found.";

        featureList.appendChild(item);

    }


    // Display URLs

    if (urls) {

        let item =
            document.createElement("li");

        item.innerHTML =
            "<strong>URLs Detected:</strong> "
            + urls.join(", ");

        featureList.appendChild(item);

    }

    else {

        let item =
            document.createElement("li");

        item.innerHTML =
            "No URLs detected.";

        featureList.appendChild(item);

    }


    // Calculate classification

    if (score >= 2) {

        result.className =
            "result phishing";

        resultTitle.innerHTML =
            "⚠ PHISHING EMAIL DETECTED";

        resultText.innerHTML =
            "This email contains suspicious keywords or URLs. "
            + "Be careful before clicking any links or sharing personal information.";

    }

    else {

        result.className =
            "result safe";

        resultTitle.innerHTML =
            "✓ SAFE EMAIL";

        resultText.innerHTML =
            "No major phishing indicators were detected in this email.";

    }


    // Display risk score

    let scoreItem =
        document.createElement("li");

    scoreItem.innerHTML =
        "<strong>Risk Score:</strong> "
        + score;

    featureList.appendChild(scoreItem);

}
